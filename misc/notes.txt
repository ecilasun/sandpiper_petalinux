// Sample code to show contents of 0x18000000:

// hw/display/zynq_simple_fb.c

#include "qemu/osdep.h"
#include "hw/sysbus.h"
#include "ui/console.h"
#include "ui/pixel_ops.h"
#include "qom/object.h"

#define TYPE_ZYNQ_SIMPLE_FB "zynq-simple-fb"
#define ZYNQ_FB_WIDTH  640
#define ZYNQ_FB_HEIGHT 480
#define ZYNQ_FB_BPP    16

typedef struct ZynqSimpleFBState {
    SysBusDevice parent_obj;
    
    MemoryRegion vram;
    QemuConsole *console;
    uint8_t *framebuffer;
    
    uint32_t width;
    uint32_t height;
    uint32_t bpp;
} ZynqSimpleFBState;

DECLARE_INSTANCE_CHECKER(ZynqSimpleFBState, ZYNQ_SIMPLE_FB, TYPE_ZYNQ_SIMPLE_FB)

static void zynq_simple_fb_update(void *opaque)
{
    ZynqSimpleFBState *s = ZYNQ_SIMPLE_FB(opaque);
    DisplaySurface *surface = qemu_console_surface(s->console);
    
    /* Convert RGB565 framebuffer to display surface format */
    uint16_t *src = (uint16_t *)s->framebuffer;
    uint32_t *dst = (uint32_t *)surface_data(surface);
    
    for (int y = 0; y < s->height; y++) {
        for (int x = 0; x < s->width; x++) {
            uint16_t pixel = *src++;
            /* RGB565 to RGB888 conversion */
            uint8_t r = ((pixel >> 11) & 0x1F) << 3;
            uint8_t g = ((pixel >> 5) & 0x3F) << 2;
            uint8_t b = (pixel & 0x1F) << 3;
            *dst++ = rgb_to_pixel32(r, g, b);
        }
    }
    
    dpy_gfx_update(s->console, 0, 0, s->width, s->height);
}

static const GraphicHwOps zynq_simple_fb_ops = {
    .gfx_update = zynq_simple_fb_update,
};

static void zynq_simple_fb_realize(DeviceState *dev, Error **errp)
{
    ZynqSimpleFBState *s = ZYNQ_SIMPLE_FB(dev);
    
    s->width = ZYNQ_FB_WIDTH;
    s->height = ZYNQ_FB_HEIGHT;
    s->bpp = ZYNQ_FB_BPP;
    
    /* Allocate VRAM as a memory region */
    memory_region_init_ram(&s->vram, OBJECT(dev), "zynq-fb.vram",
                           s->width * s->height * (s->bpp / 8), errp);
    s->framebuffer = memory_region_get_ram_ptr(&s->vram);
    
    sysbus_init_mmio(SYS_BUS_DEVICE(dev), &s->vram);
    
    /* Create the display console */
    s->console = graphic_console_init(dev, 0, &zynq_simple_fb_ops, s);
    qemu_console_resize(s->console, s->width, s->height);
}

static void zynq_simple_fb_class_init(ObjectClass *klass, void *data)
{
    DeviceClass *dc = DEVICE_CLASS(klass);
    dc->realize = zynq_simple_fb_realize;
    set_bit(DEVICE_CATEGORY_DISPLAY, dc->categories);
}

static const TypeInfo zynq_simple_fb_info = {
    .name          = TYPE_ZYNQ_SIMPLE_FB,
    .parent        = TYPE_SYS_BUS_DEVICE,
    .instance_size = sizeof(ZynqSimpleFBState),
    .class_init    = zynq_simple_fb_class_init,
};

static void zynq_simple_fb_register_types(void)
{
    type_register_static(&zynq_simple_fb_info);
}

type_init(zynq_simple_fb_register_types)



// To access this from code:
/* In zynq_init() function */
DeviceState *fb = qdev_new("zynq-simple-fb");
SysBusDevice *fb_sbd = SYS_BUS_DEVICE(fb);
sysbus_realize_and_unref(fb_sbd, &error_fatal);
sysbus_mmio_map(fb_sbd, 0, 0x18000000);
